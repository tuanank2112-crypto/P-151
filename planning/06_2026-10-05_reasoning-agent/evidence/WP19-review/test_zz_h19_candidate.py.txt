"""H19 reviewer's CANDIDATE tests (sandbox only, not in the repo): quotes that share NO word with the report."""

from src.maintenance.catalog import load_lexicon
from src.maintenance.safety import LLMSafetyAssessment, SignalJudgement, assess, merge_llm
from tests.test_maintenance.helpers import MONDAY_9AM, RESIDENT, Clock, agent_engine, decision
from tests.test_maintenance.test_p10_catalog_not_quote import FULL, T1, T3, VS, receive, reply
from src.maintenance.agent.client import ClientReply

LEX = load_lexicon().safety


def test_location_quote_sharing_no_word_is_kept():
    case = receive(T1, reply([VS], {**FULL, "quote": "the walkway by the lift"}))
    assert case.agent_decision.rejected == [] and case.ticket.status == "ROUTED"


def test_job_quote_sharing_no_word_is_kept():
    case = receive(T3, reply([dict(VS, quote="rubbish everywhere")]))
    assert case.agent_decision.valid_job_indexes == [0] and case.ticket.status == "ROUTED"


def test_invented_place_exact_rejected():
    def invented(system, user, data):
        return ClientReply(
            text=decision(**{"location.building": "S2", "location.floor": "7", "location.quote": ".."}), latency_ms=0.0
        )

    case = agent_engine(Clock(MONDAY_9AM), invented).receive("Hành lang nhiều rác, đèn cũng không sáng", resident_id=RESIDENT)
    assert case.agent_decision.rejected == ["jobs[1].asset_id"]


def test_merge_llm_output_does_not_depend_on_quote_wording():
    text = "Hành lang tầng 3 tòa S1 nhiều rác, nghe tiếng xì xì từ bình gas"
    rule = assess(text, LEX)
    a = merge_llm(rule, LLMSafetyAssessment(judgements=[SignalJudgement(signal_id="S-10", status="PRESENT", quote="tiếng xì xì từ bình gas")]), text, LEX)
    b = merge_llm(rule, LLMSafetyAssessment(judgements=[SignalJudgement(signal_id="S-10", status="PRESENT", quote="gas leak")]), text, LEX)
    assert (a.risk_level, a.notes, [(s.signal_id, s.status, s.rule_id) for s in a.signals]) == (
        b.risk_level, b.notes, [(s.signal_id, s.status, s.rule_id) for s in b.signals]
    )

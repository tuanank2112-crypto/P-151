def first_sentence(text: str, limit: int = 120) -> str:
    stripped = text.strip()
    first = re.split(r"(?<=[.!?\n])\s+", stripped, maxsplit=1)[0].strip()
    return (first or stripped)[:limit]


def merge_llm(rule, llm_result, text, lexicon, *, model_label="llm"):
    signals = list(rule.signals)
    notes = list(rule.notes)
    known = {s.signal_id for s in lexicon.signals if s.phase in lexicon.active_phases}
    for judgement in llm_result.judgements:
        sid = judgement.signal_id
        if sid not in known:
            notes.append(f"{sid}: LLM nêu tín hiệu ngoài từ vựng P0; bỏ qua")
            continue
        status = SignalStatus(judgement.status)
        if status == SignalStatus.ABSENT:
            notes.append(f"{sid}: LLM ABSENT không bao giờ hạ mức luật; bỏ qua")
            continue
        quote = (judgement.quote or "").strip()[:300]
        if not quote:
            notes.append(f"{sid}: LLM {status} không kèm trích dẫn; vẫn tính (trích dẫn chỉ để audit)")
        best_rule = max((safety.SEVERITY[s.status] for s in signals if s.signal_id == sid), default=-1)
        if safety.SEVERITY[status] > best_rule:
            signals.append(
                RiskSignal(
                    signal_id=sid,
                    status=status,
                    source_quote=quote or first_sentence(text) or "(trống)",
                    detected_by=DetectedBy.LLM,
                    rule_id="LLM" if quote else "LLM-NOQUOTE",
                )
            )
        else:
            notes.append(f"{sid}: LLM {judgement.status} không cao hơn luật; giữ luật")
    merged = safety._dedupe_signals(signals)
    return safety.SafetyAssessment(
        risk_level=safety.derive_risk_level(merged, lexicon),
        signals=merged,
        notes=tuple(notes),
        proposer=f"rules+llm:{model_label}",
    )

"""WP18 package check: every Markdown link and every repo path in backticks of the given files must exist.

Usage (repo root): python check_links.py <file.md> ...   Exit 0 iff nothing is missing. Prints one line per target.
Markdown links resolve against the file's folder; backtick paths that start with a known top folder resolve against the
repo root, and plan-relative ones (specs/, handoffs/, reports/, evidence/) against the plan folder.
"""

import re
import sys
from pathlib import Path

ROOT = Path.cwd()
PLAN = ROOT / "planning" / "06_2026-10-05_reasoning-agent"
LINK = re.compile(r"\]\(([^)\s]+)\)")
TICK = re.compile(r"`([^`\s]+)`")

PLAN_REL = ("specs/", "handoffs/", "reports/", "evidence/")
FUTURE = ("evidence/WP19", "reports/R18_thi-cong_WP19.md", "tests/test_maintenance/test_p10_catalog_not_quote.py")
missing = 0
for name in sys.argv[1:]:
    path = Path(name)
    text = path.read_text(encoding="utf-8")
    targets = []
    for raw in LINK.findall(text):
        if raw.startswith(("http://", "https://", "#", "mailto:")):
            continue
        targets.append(("link", raw, (path.parent / raw.split("#", 1)[0]).resolve()))
    for raw in TICK.findall(text):
        clean = raw.rstrip(".,;:)").split("::", 1)[0]
        clean = re.sub(r":\d+([-–,]\d+)*$", "", clean)  # file.py:100-101 -> file.py
        if "*" in clean or "<" in clean or "{" in clean or "…" in clean:
            continue
        if clean.startswith(TOP):
            base = ROOT
        elif clean.startswith(PLAN_REL):
            base = PLAN
        else:
            continue
        targets.append(("path", raw, (base / clean).resolve()))
    for kind, raw, target in targets:
        rel = target.relative_to(ROOT) if target.is_relative_to(ROOT) else target
        if target.exists():
            state = "OK"
        elif any(str(rel).replace("\\", "/").startswith(f"planning/06_2026-10-05_reasoning-agent/{f}") or
                 str(rel).replace("\\", "/") == f for f in FUTURE):
            state = "FUTURE (WP19 tạo)"
        else:
            state = "MISSING"
            missing += 1
        print(f"{state:18} {name} {kind} {raw}")
print(f"MISSING={missing}")
sys.exit(1 if missing else 0)
